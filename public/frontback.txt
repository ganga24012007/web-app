index.html

<!DOCTYPE html>
<html ng-app="studentApp">
<head>
    <title>Student Management</title>
    <script src="https://ajax.googleapis.com/ajax/libs/angularjs/1.8.2/angular.min.js"></script>
</head>

<body ng-controller="StudentController">

    <h2>Student Record Management</h2>

    <form ng-submit="saveStudent()">
        <input ng-model="name" placeholder="Name">
        <input ng-model="dept" placeholder="Department">
        <input type="number" ng-model="age" placeholder="Age">

        <button type="submit">
            {{ editId ? "Update" : "Add Student" }}
        </button>
    </form>

    <h3>Search by Name</h3>
    <input ng-model="searchText" placeholder="Enter name">

    <h3>Student List</h3>

    <table border="1" cellpadding="8">
        <tr>
            <th>Name</th>
            <th>Department</th>
            <th>Age</th>
            <th>Actions</th>
        </tr>

        <tr ng-repeat="s in students | filter:{name:searchText}">
            <td>{{s.name}}</td>
            <td>{{s.dept}}</td>
            <td>{{s.age}}</td>
            <td>
                <button ng-click="editStudent(s)">Edit</button>
                <button ng-click="deleteStudent(s._id)">Delete</button>
            </td>
        </tr>
    </table>

    <script>
        var app = angular.module("studentApp", []);

        app.controller("StudentController", function($scope, $http) {
            var url = "/students";

            $scope.students = [];
            $scope.editId = null;

            // GET
            function load() {
                $http.get(url).then(function(res) {
                    $scope.students = res.data;
                });
            }

            // ADD / UPDATE
            $scope.saveStudent = function() {
                var s = {
                    name: $scope.name,
                    dept: $scope.dept,
                    age: $scope.age
                };

                var request = $scope.editId
                    ? $http.put(url + "/" + $scope.editId, s)
                    : $http.post(url, s);

                request.then(function() {
                    $scope.name = "";
                    $scope.dept = "";
                    $scope.age = "";
                    $scope.editId = null;
                    load();
                });
            };

            // EDIT
            $scope.editStudent = function(s) {
                $scope.editId = s._id;
                $scope.name = s.name;
                $scope.dept = s.dept;
                $scope.age = s.age;
            };

            // DELETE
            $scope.deleteStudent = function(id) {
                $http.delete(url + "/" + id).then(function() {
                    load();
                });
            };

            load();
        });
    </script>

</body>
</html>

server.js

const express = require("express");
const mongoose = require("mongoose");
const cors = require("cors");

const app = express();

app.use(cors());
app.use(express.json());
app.use(express.static(__dirname));

// Connect to MongoDB
mongoose.connect("mongodb://127.0.0.1:27017/labdb")
    .then(() => console.log("MongoDB connected"))
    .catch(err => console.log(err));

// Student collection
const studentSchema = new mongoose.Schema({
    name: String,
    dept: String,
    age: Number
});

const Student = mongoose.model("Student", studentSchema);

// CREATE - Add student
app.post("/students", async (req, res) => {
    const student = await Student.create(req.body);
    res.json(student);
});

// READ - Display students
app.get("/students", async (req, res) => {
    const students = await Student.find();
    res.json(students);
});

// UPDATE - Edit student
app.put("/students/:id", async (req, res) => {
    const student = await Student.findByIdAndUpdate(
        req.params.id,
        req.body,
        { new: true }
    );
    res.json(student);
});

// DELETE - Remove student
app.delete("/students/:id", async (req, res) => {
    await Student.findByIdAndDelete(req.params.id);
    res.json({ message: "Student deleted" });
});

// Start server
mongoose.connection.once("open", () => {
    app.listen(3000, () => {
        console.log("Server running at http://localhost:3000");
    });
});
